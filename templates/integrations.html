High impact

Recruiting (ATS)

Track which campaign brought which actual hire. Bullhorn, SmartRecruiters, Workday, Successfactors.

  • UTM-Cookie wird beim Klick auf /s/<code> gesetzt (1st-party, GDPR-konform)
  • JS-Snippet auf der Application-Page liest Cookie + schreibt in hidden form-field
  • ATS speichert die utm_link_id mit jedem Bewerber
  • Per ATS-API ziehst du: Cost-per-Application, Cost-per-Hire, Time-to-Hire pro Quelle
Show JS snippet 
// Paste this into your application page <head>
(function(){
  var params = new URLSearchParams(location.search);
  var key = 'utm_link_id';
  var fromUrl = params.get('utm_id') || params.get(key);
  if (fromUrl) {
    document.cookie = key+'='+encodeURIComponent(fromUrl)+
      '; path=/; max-age=2592000; SameSite=Lax';
  }
  // Wait for form to appear, inject hidden field
  document.addEventListener('DOMContentLoaded', function(){
    var match = document.cookie.match(new RegExp('(^|;\\\\s*)'+key+'=([^;]+)'));
    if (!match) return;
    var val = decodeURIComponent(match[2]);
    document.querySelectorAll('form').forEach(function(f){
      if (f.querySelector('input[name="'+key+'"]')) return;
      var inp = document.createElement('input');
      inp.type='hidden'; inp.name=key; inp.value=val;
      f.appendChild(inp);
    });
  });
})();

Business value: directly answers "which marketing channel produced our best hires" — the question the CMO asks every quarter.

(function(){
  var p = new URLSearchParams(location.search);
  ['source','medium','campaign','term','content','id'].forEach(function(k){
    var v = p.get('utm_'+k);
    if (v) b['utm_'+k] = v;
  });
  // Adecco custom dimensions
  if (p.get('utm_id')) {
    var id = p.get('utm_id').split('-');
    b.tag_agency = id[0] || '';
    b.tag_targeting = id[1] || '';
    b.tag_segmentation = id[2] || '';
    b.tag_asset = id.slice(3).join('-');
  }
})();
